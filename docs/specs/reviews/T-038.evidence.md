## Verificação do AC-T038.1 (para reproduzir)
Rodada com `java -cp <snakeyaml-2.6.jar do .m2> VerificaWorkflow.java .github/workflows/security.yml` (esperado: `OK AC-T038.1`, exit 0). Fonte:

```java
import java.io.*;
import java.util.*;
import org.yaml.snakeyaml.Yaml;

public class VerificaWorkflow {
  @SuppressWarnings("unchecked")
  public static void main(String[] a) throws Exception {
    Map<String, Object> raiz = new Yaml().load(new FileReader(a[0]));
    Map<String, Object> jobs = (Map<String, Object>) raiz.get("jobs");
    Map<String, Object> dc = (Map<String, Object>) jobs.get("dependency-check");
    String run = null;
    for (Object s : (List<Object>) dc.get("steps")) {
      Map<String, Object> m = (Map<String, Object>) s;
      if ("Dependency-Check".equals(m.get("name"))) run = (String) m.get("run");
    }
    List<String> falhas = new ArrayList<>();
    for (String t : new String[] {"--disableOssIndex", "--failOnCVSS 7", "--nvdApiKey", "--suppression"})
      if (run == null || !run.contains(t)) falhas.add(t);
    if (run != null && run.contains("--disableRetireJS")) falhas.add("nao deve ter --disableRetireJS");
    if (falhas.isEmpty()) System.out.println("OK AC-T038.1");
    else { System.out.println("FALHA AC-T038.1: " + falhas); System.exit(1); }
  }
}
```
